private WalletRecordFilter $walletRecordFilter;
/**
* WalletRepository constructor.
* @param WalletFilter $filter
*/
public function __construct(WalletRecordFilter $walletRecordFilter ,
        WalletFilter $walletFilter)
{
$this->walletFilter = $walletFilter;
$this->walletRecordFilter = $walletRecordFilter;
}
function index()
{
$walletRecord = WalletRecord::filter($this->walletRecordFilter
            )->orderBy("created_at", "desc")->paginate(request("per_page") ?? 5
            );
$wallet = Wallet::filter($this->walletFilter)->orderBy("created_at",
            "desc")->paginate(request("per_page") ?? 5);
$data = collect([...$walletRecord , ...$wallet ])->sortBy("created_at"
            )->all();
$arr = [] ;
return ($data);
$canWithdraw = auth()->user()->wallets()->whereStatus("pending")->count
            () == 0 && auth()->user()->balance > 0;
$response = [
"balance" => auth()->user()->balance,
"canWithdraw" => $canWithdraw,
"records" => getPaginates(WalletRecordResource::collection($data
                ))
];
return sendResponse($response);
}
function show($id)
{
$model = WalletRecord::findOrFail($id);
$check = $model->provider->is(auth()->user()) ;
if(!$check){
$message = __("messages.u_r_not_permited");
return sendError($message, ['not authrized' => $message]);
}
return sendResponse(new WalletRecordResource($model));
}
